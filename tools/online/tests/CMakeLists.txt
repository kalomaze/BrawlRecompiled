cmake_minimum_required(VERSION 3.20)
project(BrawlOnlineTransportTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../../wiicompiled/runtime" ABSOLUTE)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
if(NOT EXISTS "${GEKKONET_INCLUDE}/gekkonet.h")
    message(FATAL_ERROR "Set GEKKONET_INCLUDE to Brawl Online's GekkoLib/include directory")
endif()
add_executable(control_socket_tests "${RUNTIME}/tests/local_control_socket_tests.cpp")
add_executable(rollback_udp_tests "${RUNTIME}/tests/brawl_rollback_udp_tests.cpp" "${RUNTIME}/src/brawl_rollback_udp.cpp")
foreach(name brawl_snapshot_store brawl_write_watch brawl_dirty_memory)
    add_executable(${name}_tests "${RUNTIME}/tests/${name}_tests.cpp")
endforeach()
foreach(target control_socket_tests rollback_udp_tests brawl_snapshot_store_tests brawl_write_watch_tests brawl_dirty_memory_tests)
    target_include_directories(${target} PRIVATE "${RUNTIME}/include" "${GEKKONET_INCLUDE}")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(WIN32)
        target_link_libraries(${target} PRIVATE ws2_32)
        target_compile_definitions(${target} PRIVATE GEKKONET_STATIC)
    endif()
    # Contract tests use assertions in every build configuration.
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
enable_testing()
add_test(NAME udp_transport COMMAND rollback_udp_tests)
foreach(name brawl_snapshot_store brawl_write_watch brawl_dirty_memory)
    add_test(NAME ${name} COMMAND ${name}_tests)
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
endforeach()
add_test(NAME control_and_protocol COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/.." -p "test_*.py" -v)
set_tests_properties(control_and_protocol PROPERTIES ENVIRONMENT "BRAWL_CONTROL_TEST_SERVER=$<TARGET_FILE:control_socket_tests>" TIMEOUT 60)
set_tests_properties(udp_transport PROPERTIES TIMEOUT 10)
